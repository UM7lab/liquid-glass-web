cmake_minimum_required(VERSION 3.25)
project(glass_reference LANGUAGES CXX)
set(NATIVE_ROOT "" CACHE PATH "Path to Liquid Glass Windows revision 3.2")
add_executable(glass_reference reference.cpp)
target_include_directories(glass_reference PRIVATE "${NATIVE_ROOT}/src" "${NATIVE_ROOT}/include" "${NATIVE_ROOT}/shaders")
target_compile_features(glass_reference PRIVATE cxx_std_17)
target_compile_options(glass_reference PRIVATE /permissive /await /EHsc /utf-8)
target_compile_definitions(glass_reference PRIVATE LG_GLASS_EXPORTS UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(glass_reference PRIVATE d3d11 dxgi dcomp d3dcompiler dwmapi user32 gdi32 windowsapp)
